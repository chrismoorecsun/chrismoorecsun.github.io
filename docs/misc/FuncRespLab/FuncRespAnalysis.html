      lm_gen <- lm(dat$rem ~ dat$init + 0)
      # Save parameters

      # Create x values
      x_vec <- seq(from = 1, to = 50)
      # Predict y for the x values in the linear equation
      predict_y <- coeff_gen*x_vec
      # Add liner model to plot
      lines(x = x_vec, y = predict_y, col = "tomato", lwd = 2)
    
    # Fit non-linear (type II FR) model
      # Fit parameters
      nls_gen <- nls(rem ~ a*init / (1 + a*b*init), data = dat, start = c(a = 0.1, b = 1))
      # Save parameters
      coeffsNLS_gen <- coefficients(nls_gen)
      # Create x values (again, but not necessary)
      x_vec <- seq(from = 1, to = 50)
      # Predict y for the x values in the non-linear equation
      predNLS_y <- coeffsNLS_gen["a"]*x_vec / (1 + coeffsNLS_gen["a"]*coeffsNLS_gen["b"]*x_vec)
      # Add non-liner model to plot
      lines(x = x_vec, y = predNLS_y, col = "skyblue", lwd = 2)

  # Statistically compare models
  AIC(lm_gen, nls_gen)
